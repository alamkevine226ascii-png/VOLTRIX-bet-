   119→   119→function Pitch({ team }: { team: TeamLineupsDTO }) {
   120→   120→  const rows: Record<string, PlayerLineupsDTO[]> = { G: [], D: [], M: [], F: [] };
   121→   121→  team.starters.forEach((p) => {
   122→   122→    const k = p.position ?? "M";
   123→   123→    if (!rows[k]) rows[k] = [];
   124→   124→    rows[k].push(p);
   125→   125→  });
   126→   126→  const order = ["F", "M", "D", "G"];
   127→   127→  return (
   128→   128→    <div className="rounded-3xl bg-pitch p-3 ring-1 ring-white/10">
   129→   129→      <div className="relative flex aspect-[3/4] w-full flex-col justify-between rounded-2xl border-2 border-pitch-line bg-[linear-gradient(rgba(255,255,255,0.045)_1px,transparent_1px)] bg-[length:100%_12.5%]">
   130→   130→        {/* lignes du terrain */}
   131→   131→        <div className="absolute inset-x-0 top-1/2 h-[2px] bg-pitch-line" />
   132→   132→        <div className="absolute left-1/2 top-1/2 h-14 w-14 -translate-x-1/2 -translate-y-1/2 rounded-full border-2 border-pitch-line" />
   133→   133→        <div className="absolute inset-x-[22%] top-0 h-10 border-x-2 border-b-2 border-pitch-line" />
   134→   134→        <div className="absolute inset-x-[22%] bottom-0 h-10 border-x-2 border-t-2 border-pitch-line" />
   135→   135→
   136→   136→        {order.map((k) => {
   137→   137→          const players = rows[k];
   138→   138→          if (players.length === 0) return null;
   139→   139→          return (
   140→   140→            <div key={k} className="relative z-10 flex items-start justify-center gap-1 px-2">
   141→   141→              {players.map((p) => (
   142→   142→                <div key={p.athleteId} className="flex w-[52px] flex-col items-center">
   143→   143→                  <span className="flex h-8 w-8 items-center justify-center rounded-full bg-neon text-[11px] font-black text-black ring-2 ring-black/30">
   144→   144→                    {p.jersey ?? "?"}
   145→   145→                  </span>
   146→   146→                  <span className="mt-1 max-w-full truncate rounded bg-black/60 px-1 py-[1px] text-[7.5px] font-bold uppercase tracking-wide text-white/85">
   147→   147→                    {p.shortName ?? p.name}
   148→   148→                  </span>
   149→   149→                </div>
   150→   150→              ))}
   151→   151→            </div>
   152→   152→          );
   153→   153→        })}
   154→   154→      </div>
   155→   155→    </div>
   156→   156→  );
   157→   157→}
