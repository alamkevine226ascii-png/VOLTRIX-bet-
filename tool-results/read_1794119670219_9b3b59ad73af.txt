    83→/* ---------- Terrain (Compos) ---------- */
    84→function Pitch({ team }: { team: TeamLineupsDTO }) {
    85→  const rows: Record<string, PlayerLineupsDTO[]> = { G: [], D: [], M: [], F: [] };
    86→  team.starters.forEach((p) => {
    87→    const k = p.position ?? "M";
    88→    if (!rows[k]) rows[k] = [];
    89→    rows[k].push(p);
    90→  });
    91→  const order = ["F", "M", "D", "G"];
    92→  return (
    93→    <div className="rounded-3xl bg-pitch p-3 ring-1 ring-white/10">
    94→      <div className="relative flex aspect-[3/4] w-full flex-col justify-between rounded-2xl border-2 border-pitch-line bg-[linear-gradient(rgba(255,255,255,0.045)_1px,transparent_1px)] bg-[length:100%_12.5%]">
    95→        <div className="absolute inset-x-0 top-1/2 h-[2px] bg-pitch-line" />
    96→        <div className="absolute left-1/2 top-1/2 h-14 w-14 -translate-x-1/2 -translate-y-1/2 rounded-full border-2 border-pitch-line" />
    97→        <div className="absolute inset-x-[22%] top-0 h-10 border-x-2 border-b-2 border-pitch-line" />
    98→        <div className="absolute inset-x-[22%] bottom-0 h-10 border-x-2 border-t-2 border-pitch-line" />
    99→
   100→        {order.map((k) => {
   101→          const players = rows[k];
   102→          if (players.length === 0) return null;
   103→          return (
   104→            <div key={k} className="relative z-10 flex items-start justify-center gap-1 px-2">
   105→              {players.map((p) => (
   106→                <div key={p.athleteId} className="flex w-[52px] flex-col items-center">
   107→                  <span className="flex h-8 w-8 items-center justify-center rounded-full bg-neon text-[11px] font-black text-black ring-2 ring-black/30">
   108→                    {p.jersey ?? "?"}
   109→                  </span>
   110→                  <span className="mt-1 max-w-full truncate rounded bg-black/60 px-1 py-[1px] text-[7.5px] font-bold uppercase tracking-wide text-white/85">
   111→                    {p.shortName ?? p.name}
   112→                  </span>
   113→                </div>
   114→              ))}
   115→            </div>
   116→          );
   117→        })}
   118→      </div>
   119→    </div>
   120→  );
   121→}

   123→/* ---------- Écran ---------- */
   124→export function MatchScreen({
   125→  matchId,
   126→  leagueCode,
   127→  date,
   128→  light,
   129→  onBack,
   130→  onTicket,
   131→}: {
   132→  matchId: string;
   133→  leagueCode: string;
   134→  date: string;
   135→  light?: LightMatch;
   136→  onBack: () => void;
   137→  onTicket: (draft: SelectionDraft) => void;
   138→}) {
   139→  const [detail, setDetail] = useState<MatchDetailDTO | null>(null);
   140→  const [loading, setLoading] = useState(true);
   141→  const [error, setError] = useState<string | null>(null);
   142→  const [tab, setTab] = useState<TabId>("synthese");
   143→  const [lineups, setLineups] = useState<Awaited<ReturnType<typeof loadLineups>>>(null);
   144→  const [lineupsLoading, setLineupsLoading] = useState(false);
   145→  const isLive = getApiBase().length > 0;
   146→
   147→  const load = useCallback(async () => {
   148→    setLoading(true);
   149→    setError(null);
   150→    try {
   151→      const res = isLive ? await fetchMatchDetail(matchId, leagueCode, date) : demoMatchDetail(matchId);
   152→      setDetail(res);
   153→    } catch (e) {
   154→      setError(e instanceof Error ? e.message : "Analyse indisponible");
   155→    } finally {
   156→      setLoading(false);
   157→    }
   158→  }, [matchId, leagueCode, date, isLive]);
   159→
   160→  useEffect(() => {
   161→    void load();
   162→  }, [load]);
   163→
   164→  async function loadLineups(): Promise<ReturnType<typeof demoLineups> | null> {
   165→    if (isLive) return fetchLineups(matchId, leagueCode);
   166→    return demoLineups(matchId);
   167→  }
   168→
   169→  useEffect(() => {
   170→    if (tab !== "compos" || lineups || lineupsLoading) return;
   171→    let cancelled = false;
   172→    setLineupsLoading(true);
   173→    loadLineups()
   174→      .then((r) => {
   175→        if (!cancelled) setLineups(r);
   176→      })
   177→      .catch(() => {
   178→        if (!cancelled) setLineups(null);
   179→      })
   180→      .finally(() => {
   181→        if (!cancelled) setLineupsLoading(false);
   182→      });
   183→    return () => {
   184→      cancelled = true;
   185→    };
   186→  }, [tab]);
   187→
   188→  /* Onglet Compos : équipe affichée */
   189→  const [composSide, setComposSide] = useState<"home" | "away">("home");
   190→
   191→  const pred = detail?.prediction;
   192→  const bestPick = useMemo(() => pred?.recommendedBets[0] ?? null, [pred]);
   193→
   194→  const live = detail?.status === "in" || light?.status === "in";
   195→  const hasScore = detail && detail.homeScore != null && detail.awayScore != null;
   196→
   197→  const homeName = detail?.home.name ?? light?.home.name ?? "?";
   198→  const awayName = detail?.away.name ?? light?.away.name ?? "?";
   199→  const homeShort = (light?.home.shortName ?? detail?.home.name ?? "Domicile").slice(0, 10);
   200→  const awayShort = (light?.away.shortName ?? detail?.away.name ?? "Extérieur").slice(0, 10);
   201→
   202→  /* Segments 1X2 pour la barre unique */
   203→  const segments = useMemo(() => {
   204→    if (!pred) return [];
   205→    return [
   206→      {
   207→        id: "home" as const,
   208→        value: pred.probs.home,
   209→        label: homeShort,
   210→        odds: light?.mlHome ?? detail?.odds?.moneyline.home.close ?? null,
   211→      },
   212→      {
   213→        id: "draw" as const,
   214→        value: pred.probs.draw,
   215→        label: "Nul",
   216→        odds: light?.mlDraw ?? detail?.odds?.moneyline.draw.close ?? null,
   217→      },
   218→      {
   219→        id: "away" as const,
   220→        value: pred.probs.away,
   221→        label: awayShort,
   222→        odds: light?.mlAway ?? detail?.odds?.moneyline.away.close ?? null,
   223→      },
   224→    ];
   225→  }, [pred, light, detail, homeShort, awayShort]);
   226→
   227→  /* Phrase de synthèse en français clair */
   228→  const summary = useMemo(() => {
   229→    if (!pred || !detail) return null;
   230→    const fav = pred.probs.home >= pred.probs.away
   231→      ? { name: detail.home.name, p: pred.probs.home }
   232→      : { name: detail.away.name, p: pred.probs.away };
   233→    const top = pred.topScores[0];
   234→    const parts = [
   235→      `Favori : ${fav.name} (${pct(fav.p)})`,
   236→      `${pred.lambda.total.toFixed(1)} buts attendus`,
   237→      top ? `score le plus probable ${top.score}` : null,
   238→    ];
   239→    if (pred.valueBets.length > 0) {
   240→      parts.push(`${pred.valueBets.length} value bet${pred.valueBets.length > 1 ? "s" : ""} détectée${pred.valueBets.length > 1 ? "s" : ""}`);
   241→    }
   242→    return parts.filter(Boolean).join(" · ");
   243→  }, [pred, detail]);
   244→
   245→  const openTicketFor = (id: "home" | "draw" | "away") => {
   246→    if (!detail || !pred) return;
   247→    const pick1x2 =
   248→      id === "home" ? `1 - ${detail.home.name}` : id === "away" ? `2 - ${detail.away.name}` : "Match nul";
   249→    onTicket({
   250→      matchId,
   251→      leagueCode,
   252→      matchDate: detail.matchDate,
   253→      market: "1X2",
   254→      pick: pick1x2,
   255→      prob: id === "home" ? pred.probs.home : id === "away" ? pred.probs.away : pred.probs.draw,
   256→      odds:
   257→        id === "home"
   258→          ? light?.mlHome ?? detail.odds?.moneyline.home.close ?? null
   259→          : id === "away"
   260→            ? light?.mlAway ?? detail.odds?.moneyline.away.close ?? null
   261→            : light?.mlDraw ?? detail.odds?.moneyline.draw.close ?? null,
   262→      homeName: detail.home.name,
   263→      awayName: detail.away.name,
   264→      homeLogo: light?.home.logo ?? null,
   265→      awayLogo: light?.away.logo ?? null,
   266→    });
   267→  };
   268→
   269→  const openTicketMarket = (market: string, pick: string, prob: number, odds: number | null) => {
   270→    if (!detail) return;
   271→    onTicket({
   272→      matchId,
   273→      leagueCode,
   274→      matchDate: detail.matchDate,
   275→      market,
   276→      pick,
   277→      prob,
   278→      odds,
   279→      homeName: detail.home.name,
   280→      awayName: detail.away.name,
   281→      homeLogo: light?.home.logo ?? null,
   282→      awayLogo: light?.away.logo ?? null,
   283→    });
   284→  };
   285→
   286→  const teamBlock = (t: TeamAnalysisDTO) => (
   287→    <div className="rounded-2xl bg-ink-3 p-3.5">
   288→      <div className="flex items-center gap-2.5">
   289→        <TeamBadge name={t.name} logo={t.logo} size={38} />
   290→        <div className="min-w-0">
   291→          <p className="truncate text-[12px] font-black uppercase tracking-wide text-white">{t.name}</p>
   292→          <p className="text-[9.5px] font-bold text-white/40">
   293→            Elo {t.elo} {t.rank ? `· ${t.rank}e` : ""} {t.points ? `· ${t.points} pts` : ""}
   294→          </p>
   295→        </div>
   296→      </div>
   297→      <div className="mt-2.5 grid grid-cols-2 gap-x-3 gap-y-1.5 text-[9.5px]">
   298→        {[
   299→          ["Forme", `${Math.round(t.formScore * 100)}/100`],
   300→          ["Buts/match", t.goalsForPerMatch.toFixed(2)],
   301→          ["Encaissés/match", t.goalsAgainstPerMatch.toFixed(2)],
   302→          ["Clean sheets", `${t.cleanSheetsPct}%`],
   303→          ["Sans marquer", `${t.failedToScorePct}%`],
   304→          ["Blessés (clés)", `${t.injuriesCount} (${t.injuriesKey})`],
   305→          ["Matchs/14 j", String(t.matchesLast14Days)],
   306→          ["Dernier match", t.fatigueDaysSinceLast != null ? `il y a ${t.fatigueDaysSinceLast} j` : "—"],
   307→        ].map(([k, v]) => (
   308→          <div key={k} className="flex items-baseline justify-between gap-1 border-b border-white/5 pb-1">
   309→            <span className="text-white/40">{k}</span>
   310→            <span className="font-bold text-white/85">{v}</span>
   311→          </div>
   312→        ))}
   313→      </div>
   314→      <div className="mt-2 flex items-center justify-between">
   315→        <span className="text-[8.5px] font-bold uppercase tracking-widest text-white/35">Forme</span>
   316→        <FormDots form={t.form.map((f) => f.result).join("")} />
   317→      </div>
   318→    </div>
   319→  );
   320→
   321→  return (
   322→    <div className="flex h-full flex-col bg-ink">
   323→      {/* ============ HÉRO : image du stade + voile sombre ============ */}
   324→      <div className="relative shrink-0">
   325→        <div className="absolute inset-0 overflow-hidden" aria-hidden="true">
   326→          <img
   327→            src="/hero-match.jpg"
   328→            alt=""
   329→            className="h-full w-full object-cover object-center"
   330→          />
   331→          <div className="absolute inset-0 bg-gradient-to-b from-ink/75 via-ink/40 to-ink" />
   332→        </div>
   333→
   334→        {/* barre supérieure */}
   335→        <header className="relative flex items-center justify-between px-5 pt-6">
   336→          <button
   337→            onClick={onBack}
   338→            className="flex h-9 w-9 items-center justify-center rounded-full bg-ink-3/90 text-white backdrop-blur transition-colors hover:bg-neon hover:text-black"
   339→            aria-label="Retour"
   340→          >
   341→            <ChevronLeft size={18} strokeWidth={2.4} />
   342→          </button>
   343→          <div className="flex min-w-0 items-center gap-2">
   344→            <span className="max-w-[150px] truncate text-[10px] font-black uppercase tracking-widest text-white/80">
   345→              {detail?.leagueName ?? light?.leagueName ?? leagueCode}
   346→            </span>
   347→            {live ? (
   348→              <span className="flex items-center gap-1 rounded-full bg-neon px-2.5 py-1 text-[9px] font-black uppercase text-black">
   349→                <span className="live-dot h-1.5 w-1.5 rounded-full bg-black" />
   350→                Live
   351→              </span>
   352→            ) : (
   353→              <span className="rounded-full bg-ink-3/90 px-2.5 py-1 text-[9px] font-black uppercase tracking-wide text-white/70 backdrop-blur">
   354→                {detail?.status === "post" ? "Terminé" : "À venir"}
   355→              </span>
   356→            )}
   357→          </div>
   358→          <button
   359→            onClick={() => void load()}
   360→            className="flex h-9 w-9 items-center justify-center rounded-full bg-ink-3/90 text-white backdrop-blur transition-colors hover:bg-neon hover:text-black"
   361→            aria-label="Réanalyser"
   362→          >
   363→            <RefreshCw size={15} strokeWidth={2.2} className={loading ? "animate-spin" : ""} />
   364→          </button>
   365→        </header>
   366→
   367→        {/* équipes + score */}
   368→        <div className="relative mt-2 grid grid-cols-[1fr_auto_1fr] items-center px-6 pb-4 pt-1">
   369→          <div className="flex flex-col items-center gap-1.5">
   370→            <TeamBadge name={homeName} logo={light?.home.logo ?? detail?.home.logo ?? null} size={56} />
   371→            <span className="max-w-[104px] text-center text-[8.5px] font-extrabold leading-[1.3] tracking-wide text-white">
   372→              {homeName.toUpperCase()}
   373→            </span>
   374→          </div>
   375→          <div className="mx-3 min-w-[88px] px-1 text-center">
   376→            {hasScore ? (
   377→              <>
   378→                <span className="font-display text-[36px] leading-none tracking-wide text-white">
   379→                  {detail!.homeScore}:{detail!.awayScore}
   380→                </span>
   381→                <span className="mt-1 block text-[8.5px] font-semibold uppercase tracking-widest text-white/50">
   382→                  {detail?.statusDetail || "FT"}
   383→                </span>
   384→              </>
   385→            ) : (
   386→              <>
   387→                <span className="font-display text-[28px] leading-none tracking-wide text-white">
   388→                  {detail ? timeLabel(detail.matchDate) : light ? timeLabel(light.date) : "--:--"}
   389→                </span>
   390→                <span className="mt-1 block text-[8.5px] font-semibold uppercase tracking-widest text-white/50">
   391→                  Coup d&apos;envoi
   392→                </span>
   393→              </>
   394→            )}
   395→          </div>
   396→          <div className="flex flex-col items-center gap-1.5">
   397→            <TeamBadge name={awayName} logo={light?.away.logo ?? detail?.away.logo ?? null} size={56} />
   398→            <span className="max-w-[104px] text-center text-[8.5px] font-extrabold leading-[1.3] tracking-wide text-white">
   399→              {awayName.toUpperCase()}
   400→            </span>
   401→          </div>
   402→        </div>
   403→
   404→        {/* lieu */}
   405→        <div className="relative flex justify-center pb-3">
   406→          <span className="inline-flex max-w-[85%] items-center gap-1.5 rounded-full bg-ink-3/80 px-3 py-1 text-[9px] font-bold uppercase tracking-wide text-white/70 backdrop-blur">
   407→            <MapPin size={10} strokeWidth={2.4} className="shrink-0 text-neon" />
   408→            <span className="truncate">
   409→              {detail?.venue.name ?? light?.venue.name ?? "Stade"}
   410→              {detail?.venue.city ? ` · ${detail.venue.city}` : ""}
   411→            </span>
   412→          </span>
   413→        </div>
   414→      </div>
   415→
   416→      {/* ============ CONTENU ============ */}
   417→      {loading ? (
   418→        <LoadingPanel label="Le moteur VOLTRIX analyse le match…" />
   419→      ) : error || !detail || !pred ? (
   420→        <ErrorPanel message={error ?? "Analyse indisponible"} onRetry={() => void load()} isDemo={!isLive} />
   421→      ) : (
   422→        <main className="no-scrollbar flex-1 overflow-y-auto px-5 pb-32">
   423→          {/* ---------- PRONO DU MOTEUR : une carte, une hiérarchie ---------- */}
   424→          <section className="rounded-[26px] bg-neon px-5 pb-4 pt-4 text-black">
   425→            <div className="flex items-center justify-between gap-2">
   426→              <p className="text-[9px] font-black uppercase tracking-[0.2em] text-black/50">Prono du moteur</p>
   427→              <span className="flex items-center gap-1.5 text-[9px] font-black uppercase tracking-wider text-black/70">
   428→                <span title={`Confiance ${pred.confidence}/5`}>{confidenceStars(pred.confidence)}</span>
   429→                {pred.confidenceLabel}
   430→              </span>
   431→            </div>
   432→
   433→            <p className="mt-1 font-display text-[25px] leading-tight tracking-wide text-black">
   434→              {bestPick ? bestPick.pick.toUpperCase() : "—"}
   435→            </p>
   436→            <p className="mt-0.5 text-[10.5px] font-bold text-black/60">
   437→              {bestPick ? `${bestPick.market} · probabilité ${pct(bestPick.prob)}` : ""}
   438→            </p>
   439→
   440→            {/* barre 1X2 segmentée : largeur = probabilité, tap = ticket */}
   441→            <div
   442→              className="mt-3.5 flex h-[46px] overflow-hidden rounded-2xl bg-black/15"
   443→              role="group"
   444→              aria-label="Probabilités 1X2 — appuyez pour créer un ticket"
   445→            >
   446→              {segments.map((s, i) => {
   447→                const best = s.value >= Math.max(...segments.map((x) => x.value));
   448→                return (
   449→                  <button
   450→                    key={s.id}
   451→                    onClick={() => openTicketFor(s.id)}
   452→                    aria-label={`${s.label} ${Math.round(s.value * 100)} %${s.odds != null ? `, cote ${s.odds.toFixed(2)}` : ""} — créer un ticket`}
   453→                    className={`flex min-w-0 flex-col items-center justify-center gap-[3px] border-black/10 px-1 transition-transform active:scale-95 ${
   454→                      i > 0 ? "border-l" : ""
   455→                    } ${best ? "bg-black text-neon" : "text-black/75"}`}
   456→                    style={{ flexGrow: Math.max(0.09, s.value) }}
   457→                  >
   458→                    <span className="w-full truncate text-center text-[8px] font-bold uppercase tracking-wider opacity-70">
   459→                      {s.odds != null ? `${s.label} ${s.odds.toFixed(2)}` : s.label}
   460→                    </span>
   461→                    <span className="text-[14px] font-black leading-none">{Math.round(s.value * 100)}%</span>
   462→                  </button>
   463→                );
   464→              })}
   465→            </div>
   466→
   467→            {summary ? (
   468→              <p className="mt-2.5 text-[10px] font-semibold leading-relaxed text-black/65">{summary}</p>
   469→            ) : null}
   470→          </section>
   471→
   472→          {/* ---------- onglets ---------- */}
   473→          <div className="no-scrollbar sticky top-0 z-20 -mx-5 mt-4 flex gap-1.5 overflow-x-auto bg-ink/95 px-5 py-2 backdrop-blur">
