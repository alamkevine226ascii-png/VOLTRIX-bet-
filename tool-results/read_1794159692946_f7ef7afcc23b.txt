    23→/* ---------- PRNG déterministe ---------- */
    24→function seedFrom(s: string): () => number {
    25→  let h = 2166136261;
    26→  for (let i = 0; i < s.length; i++) {
    27→    h ^= s.charCodeAt(i);
    28→    h = Math.imul(h, 16777619);
    29→  }
    30→  return () => {
    31→    h ^= h << 13;
    32→    h ^= h >>> 17;
    33→    h ^= h << 5;
    34→    return ((h >>> 0) % 10_000) / 10_000;
    35→  };
    36→}
    37→
    38→const pick = <T,>(rng: () => number, arr: T[]): T => arr[Math.floor(rng() * arr.length)];
    39→const between = (rng: () => number, lo: number, hi: number): number => lo + rng() * (hi - lo);
    40→const round2 = (x: number): number => Math.round(x * 100) / 100;
    41→const round3 = (x: number): number => Math.round(x * 1000) / 1000;

   123→export function demoMatches(date: string): MatchesResponse {
   124→  const rng = seedFrom(`matches-${date}`);
   125→  const leagues: MatchesResponse["leagues"] = [];
   126→  let total = 0;
   127→
   128→  for (const code of DEMO_LEAGUES) {
   129→    const meta = LEAGUE_META[code];
   130→    const teams = TEAMS[code];
   131→    const n = 1 + Math.floor(rng() * 2); // 1-2 matchs par ligue
   132→    const matches: LightMatch[] = [];
   133→    for (let i = 0; i < n; i++) {
   134→      const h = teams[(i * 2) % teams.length];
   135→      const a = teams[(i * 2 + 1) % teams.length];
   136→      const hour = 13 + Math.floor(rng() * 9);
   137→      const minute = pick(rng, [0, 15, 30, 45]);
   138→      const kickoff = new Date(`${date}T${String(hour).padStart(2, "0")}:${String(minute).padStart(2, "0")}:00Z`);
   139→      const now = Date.now();
   140→      const ms = kickoff.getTime();
   141→      const status: LightMatch["status"] = ms > now + 30 * 60_000 ? "pre" : ms > now - 2 * 3600_000 ? "in" : "post";
   142→      const hs = status === "pre" ? null : Math.floor(rng() * 4);
   143→      const as = status === "pre" ? null : Math.floor(rng() * 3);
   144→      const mlH = round2(between(rng, 1.35, 3.4));
   145→      const mlD = round2(between(rng, 3.1, 4.2));
   146→      const mlA = round2(between(rng, 1.8, 4.6));
   147→      matches.push({
   148→        id: `demo-${date}-${code}-${i}`,
   149→        leagueCode: code,
   150→        leagueName: meta.name,
   151→        leagueShort: meta.shortName,
   152→        region: meta.region,
   153→        date: kickoff.toISOString(),
   154→        status,
   155→        statusDetail: status === "in" ? `${Math.floor(between(rng, 5, 85))}'` : status === "post" ? "FT" : "",
   156→        home: {
   157→          id: `h-${h.short}`,
   158→          name: h.name,
   159→          shortName: h.short,
   160→          logo: null,
   161→          score: hs,
   162→          form: "WDWWL".slice(0, 3 + Math.floor(rng() * 3)),
   163→          record: null,
   164→        },
   165→        away: {
   166→          id: `a-${a.short}`,
   167→          name: a.name,
   168→          shortName: a.short,
   169→          logo: null,
   170→          score: as,
   171→          form: "WDLWD".slice(0, 3 + Math.floor(rng() * 3)),
   172→          record: null,
   173→        },
   174→        hasOdds: rng() > 0.25,
   175→        oddsProvider: rng() > 0.5 ? "DraftKings" : null,
   176→        ouLine: 2.5,
   177→        mlHome: mlH,
   178→        mlDraw: mlD,
   179→        mlAway: mlA,
   180→        venue: demoVenueFor(h.short, h.name),
   181→      });
   182→      total++;
   183→    }
   184→    leagues.push({ code, name: meta.name, shortName: meta.shortName, region: meta.region, matches });
   185→  }
   186→
   187→  return { date, totalMatches: total, leagues, scanMs: 42, catalogueSize: 140 };
   188→}

   192→export function demoPredictions(matches: LightMatch[]): QuickPred[] {
   193→  return matches.map((m) => {
   194→    const rng = seedFrom(`pred-${m.id}`);
   195→    let ph = between(rng, 0.22, 0.62);
   196→    let pd = between(rng, 0.18, 0.3);
   197→    let pa = 1 - ph - pd;
   198→    if (pa < 0.08) {
   199→      pa = 0.12;
   200→      ph = Math.max(0.1, 1 - pd - pa);
   201→    }
   202→    const s = ph + pd + pa;
   203→    ph /= s;
   204→    pd /= s;
   205→    pa /= s;
   206→    const conf = 1 + Math.floor(rng() * 5);
   207→    const over = between(rng, 0.38, 0.68);
   208→    const yes = between(rng, 0.4, 0.7);
   209→    const lambdaH = round2(between(rng, 0.9, 2.1));
   210→    const lambdaA = round2(between(rng, 0.7, 1.8));
   211→    const best = ph > pa ? (ph > pd ? { market: "1X2", pick: `1 - ${m.home.name}`, prob: ph } : { market: "1X2", pick: "Match nul", prob: pd }) : pa > pd ? { market: "1X2", pick: `2 - ${m.away.name}`, prob: pa } : { market: "1X2", pick: "Match nul", prob: pd };
   212→    return {
   213→      matchId: m.id,
   214→      leagueCode: m.leagueCode,
   215→      leagueName: m.leagueName,
   216→      status: m.status,
   217→      probs: { home: round3(ph), draw: round3(pd), away: round3(pa) },
   218→      lambda: { home: lambdaH, away: lambdaA, total: round2(lambdaH + lambdaA) },
   219→      confidence: conf,
   220→      confidenceLabel: ["Faible", "Limitée", "Moyenne", "Bonne", "Élevée"][conf - 1],
   221→      recommendedBets: [
   222→        { market: best.market, pick: best.pick, prob: round3(best.prob), note: "Meilleure probabilité du moteur" },
   223→        { market: "O/U 2.5", pick: over >= 0.5 ? "Plus de 2.5" : "Moins de 2.5", prob: round3(Math.max(over, 1 - over)), note: "Distribution Poisson calibrée" },
   224→        { market: "BTTS", pick: yes >= 0.5 ? "Oui" : "Non", prob: round3(Math.max(yes, 1 - yes)), note: "Les deux équipes marquent" },
   225→      ],
   226→      overUnder: [1.5, 2.5, 3.5].map((line) => {
   227→        const o = line === 1.5 ? Math.min(0.95, over + 0.22) : line === 2.5 ? over : Math.max(0.08, over - 0.26);
   228→        return { line, over: round3(o), under: round3(1 - o) };
   229→      }),
   230→      btts: { yes: round3(yes), no: round3(1 - yes) },
   231→      topScores: [
   232→        { score: "1-1", prob: round3(between(rng, 0.09, 0.14)) },
   233→        { score: "2-1", prob: round3(between(rng, 0.07, 0.12)) },
   234→        { score: "1-0", prob: round3(between(rng, 0.07, 0.11)) },
   235→        { score: "2-0", prob: round3(between(rng, 0.06, 0.1)) },
   236→        { score: "0-1", prob: round3(between(rng, 0.05, 0.09)) },
   237→      ],
   238→      valueBetsCount: Math.floor(rng() * 3),
   239→      ouOdds: m.hasOdds ? { line: 2.5, over: round2(between(rng, 1.7, 2.1)), under: round2(between(rng, 1.7, 2.1)) } : null,
   240→    };
   241→  });
   242→}
